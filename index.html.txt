    <!-- JavaScript: funcionalidades -->

    <script>

        const coracao = document.getElementById("coracao");

        const mensagem = document.getElementById("mensagem");


        coracao.addEventListener("click", function() {

            // Faz o coração pular

            coracao.classList.remove("pular");

            void coracao.offsetWidth;

            coracao.classList.add("pular");


            // Mostra a mensagem

            mensagem.classList.add("aparecer");


            // Cria 30 corações

            for (let i = 0; i < 30; i++) {

                const novoCoracao = document.createElement("div");

                novoCoracao.classList.add("coracao-flutuante");

                novoCoracao.innerHTML = "♥";

                novoCoracao.style.left =
                    Math.random() * 100 + "vw";

                novoCoracao.style.fontSize =
                    (15 + Math.random() * 35) + "px";

                novoCoracao.style.animationDuration =
                    (2 + Math.random() * 3) + "s";

                document.body.appendChild(novoCoracao);


                // Remove após a animação

                setTimeout(function() {

                    novoCoracao.remove();

                }, 5000);

            }

        });

    </script>
